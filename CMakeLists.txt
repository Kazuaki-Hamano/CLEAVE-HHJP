  target_sources(app PRIVATE ${CMAKE_CURRENT_LIST_DIR}/src/charge_status.c)
  target_sources(app PRIVATE ${CMAKE_CURRENT_LIST_DIR}/src/app_update.c)
  execute_process(COMMAND git -c safe.directory=${CMAKE_CURRENT_LIST_DIR} rev-parse HEAD
    WORKING_DIRECTORY ${CMAKE_CURRENT_LIST_DIR}
    OUTPUT_VARIABLE CLEAVE_UPDATE_GIT_ID OUTPUT_STRIP_TRAILING_WHITESPACE)
  if(NOT CLEAVE_UPDATE_GIT_ID MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "Cannot determine CLEAVE update build identity")
  endif()
  file(GLOB_RECURSE CLEAVE_UPDATE_INPUTS RELATIVE ${CMAKE_CURRENT_LIST_DIR}
    "${CMAKE_CURRENT_LIST_DIR}/src/*" "${CMAKE_CURRENT_LIST_DIR}/config/*"
    "${CMAKE_CURRENT_LIST_DIR}/dts/*" "${CMAKE_CURRENT_LIST_DIR}/zephyr/*")
  foreach(CLEAVE_UPDATE_INPUT IN LISTS CLEAVE_UPDATE_INPUTS)
    file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/${CLEAVE_UPDATE_INPUT}" CLEAVE_INPUT_HASH)
    string(APPEND CLEAVE_UPDATE_GIT_ID "${CLEAVE_UPDATE_INPUT}:${CLEAVE_INPUT_HASH};")
  endforeach()
  file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/CMakeLists.txt" CLEAVE_CMAKE_HASH)
  string(SHA256 CLEAVE_UPDATE_HASH "${CLEAVE_UPDATE_GIT_ID};${CLEAVE_CMAKE_HASH}")
  string(SUBSTRING "${CLEAVE_UPDATE_HASH}" 0 8 CLEAVE_UPDATE_BUILD_ID)
  target_compile_definitions(app PRIVATE CLEAVE_UPDATE_BUILD_ID=\"${CLEAVE_UPDATE_BUILD_ID}\")
endif()
