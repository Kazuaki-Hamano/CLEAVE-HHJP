# Keep the pinned widget's configuration/behaviors, replacing only its LED
# renderer so both halves can show battery color while ACTIVE or IDLE.
if((CONFIG_SHIELD_CLEAVE_HHJP_LEFT OR CONFIG_SHIELD_CLEAVE_HHJP_RIGHT) AND CONFIG_RGBLED_WIDGET)
  set_property(GLOBAL PROPERTY CLEAVE_STATUS_LED_SOURCE "${CMAKE_CURRENT_LIST_DIR}/src/status_led/widget.c")
  function(cleave_replace_status_led)
    get_target_property(cleave_app_sources app SOURCES)
    set(cleave_widget_count 0)
    set(cleave_kept_sources "")
    foreach(cleave_source IN LISTS cleave_app_sources)
      if(cleave_source MATCHES "(^|/)zmk-rgbled-widget/src/widget[.]c$")
        math(EXPR cleave_widget_count "${cleave_widget_count} + 1")
      else()
        list(APPEND cleave_kept_sources "${cleave_source}")
      endif()
    endforeach()
    if(NOT cleave_widget_count EQUAL 1)
      message(FATAL_ERROR "Expected exactly one upstream LED widget source, got ${cleave_widget_count}")
    endif()
    set_property(TARGET app PROPERTY SOURCES "${cleave_kept_sources}")
    get_property(cleave_widget_source GLOBAL PROPERTY CLEAVE_STATUS_LED_SOURCE)
    target_sources(app PRIVATE "${cleave_widget_source}")
    message(STATUS "CLEAVE awake battery LED override: ${cleave_widget_source}")
  endfunction()
  cmake_language(DEFER DIRECTORY "${CMAKE_SOURCE_DIR}" CALL cleave_replace_status_led)
endif()
